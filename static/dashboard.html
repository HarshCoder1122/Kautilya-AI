JavaScript
const resp = await fetch( "https://ai.revealiq.in/api/v1/chat/completions", { method: "POST", headers: { "Authorization": "Bearer YOUR_API_KEY", "Content-Type": "application/json" }, body: JSON.stringify({ model: "kautilya-daily", messages: [ { role: "user", content: "Hello Kautilya!" } ] }) } ); const data = await resp.json(); console.log(data.choices[0].message.content);
