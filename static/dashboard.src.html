Python
import requests resp = requests.post( "https://ai.revealiq.in/api/v1/chat/completions", headers={ "Authorization": "Bearer YOUR_API_KEY", "Content-Type": "application/json" }, json={ "model": "kautilya-daily", "messages": [ {"role": "user", "content": "Hello Kautilya!"} ] } ) print(resp.json()["choices"][0]["message"]["content"])
